# Golden vectors for the WDP frame header. Read by BOTH:
#   tests/host/test_wdp.c        (C codec,      common/wdp.c)
#   client/tests/test_frame.py   (Python codec, client/wiidbg/frame.py)
# If you change the wire format you must change this file, which forces
# both implementations to be updated together.
#
# valid   <32 hex chars> <version> <kind> <seq> <cmd> <status> <flags> <len>
# invalid <32 hex chars> <error: MAGIC|KIND|FLAGS|TOO_BIG>
#
# Hex is the 16 header bytes exactly as sent on the wire.

# PING request, seq=1, 4-byte payload (worked example in docs/protocol.md)
valid   57440100000100020000000000000004 1 0 1 2 0 0 4
# PING response, seq=1, echoing 4 bytes
valid   57440101000100020000000000000004 1 1 1 2 0 0 4
# Unknown command answered with an error status and no payload
valid   57440101002a99990001000000000000 1 1 42 39321 1 0 0
# Event frame, event id 0x1001, payload of exactly the maximum size
valid   57440102000010010000000000004000 1 2 0 4097 0 0 16384
# Max-width fields (seq, cmd, status = 0xffff) to catch sign/width bugs
valid   57440101ffffffffffff000000004000 1 1 65535 65535 65535 0 16384
# Different version byte still DECODES (caller decides policy; see protocol.md)
valid   57440900000700010000000000000000 9 0 7 1 0 0 0

# --- frames a receiver must reject ---
# not our protocol at all (looks like the start of an HTTP request)
invalid 474554202f20485454502f312e310d0a MAGIC
# unknown kind (3)
invalid 57440103000100020000000000000000 KIND
# flags we do not understand
invalid 57440100000100020000000100000000 FLAGS
# payload length just over the cap (0x4001)
invalid 57440100000100020000000000004001 TOO_BIG
# payload length claiming 4 GiB: the remote-crash attempt the cap exists to stop
invalid 574401000001000200000000ffffffff TOO_BIG

# --- HELLO response payloads ---
# hello   <hex payload> <stub_major> <stub_minor> <capabilities> <max_payload> <build or - if empty>
# (decoding must give these fields; encoding those fields must reproduce the
#  LEADING bytes of the payload. It is the whole payload except where trailing
#  bytes were appended on purpose.)

# typical stub: version 0.1, no capabilities yet, 16 KiB payloads (bytes verified by hand against docs/protocol.md)
hello   0001000000000000000040000005302e312e30 0 1 0 16384 0.1.0
# empty build string, every field non-zero
hello   fffe0000deadbeef000040000000 255 254 3735928559 16384 -
# longest legal build string (31 bytes)
hello   010000000000000300001000001f78787878787878787878787878787878787878787878787878787878787878 1 0 3 4096 xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx
# trailing bytes after the build string must be ignored (future fields)
hello   0001000000000000000040000005302e312e30aabbcc 0 1 0 16384 0.1.0

# hello_invalid <hex of payload> <error: SHORT|TOO_BIG>
# only 13 bytes: the fixed part is 14
hello_invalid 00010000000000000000400000 SHORT
# build_len says 5 but only 3 bytes follow
hello_invalid 0001000000000000000040000005616263 SHORT
# build_len of 32 does not fit the 32-byte buffer (31 chars + terminator)
hello_invalid 00010000000000000000400000207979797979797979797979797979797979797979797979797979797979797979 TOO_BIG
# empty payload
hello_invalid - SHORT
